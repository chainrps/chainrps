📖 调用示例

使用 py-clob-client 的典型调用片段。

# 1) 初始化客户端并派生 API Key
from py_clob_client.client import ClobClient

client = ClobClient(
    host="https://clob.polymarket.com",
    key=PRIVATE_KEY,
    chain_id=137,
)
creds = client.create_or_derive_api_creds()

# 2) 查询市场订单簿
book = client.get_order_book(token_id)
print(book)

# 3) F3 下单 —— 创建并提交限价单
from py_clob_client.order_builder.builders import build_order

order = build_order(
    token_id=token_id,
    side="BUY",
    price=0.50,
    size=100,
)
signed = client.create_order(order)
resp = client.post_order(signed, order_type="GTC")
print(resp)

# 4) 平仓 / 取消挂单
client.cancel(order_id)
client.cancel_all()
